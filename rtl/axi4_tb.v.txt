`timescale 1ns / 1ps

module tb_axi_ram_top;

    // ----------------------------------------------------------------
    // Configuration Parameters
    // ----------------------------------------------------------------
    parameter ADDR_WIDTH = 10;   // Depth of RAM = 2^10 = 1024 locations
    parameter DATA_WIDTH = 32;   // 32-bit Data
    
    // ----------------------------------------------------------------
    // Signal Declarations
    // ----------------------------------------------------------------
    reg aclk;
    reg aresetn;
    
    // Global Error Counter for Verification
    integer error_count = 0;

    // Write Address Channel (AW)
    reg [ADDR_WIDTH-1:0]   s_axi_awaddr;
    reg                    s_axi_awvalid;
    wire                   s_axi_awready;

    // Write Data Channel (W)
    reg [DATA_WIDTH-1:0]   s_axi_wdata;
    reg [DATA_WIDTH/8-1:0] s_axi_wstrb; // Added: Byte Strobes
    reg                    s_axi_wvalid;
    wire                   s_axi_wready;

    // Write Response Channel (B)
    wire [1:0]             s_axi_bresp;
    wire                   s_axi_bvalid;
    reg                    s_axi_bready;

    // Read Address Channel (AR)
    reg [ADDR_WIDTH-1:0]   s_axi_araddr;
    reg                    s_axi_arvalid;
    wire                   s_axi_arready;

    // Read Data Channel (R)
    wire [DATA_WIDTH-1:0]  s_axi_rdata;
    wire [1:0]             s_axi_rresp;
    wire                   s_axi_rvalid;
    reg                    s_axi_rready;

    // ----------------------------------------------------------------
    // Module Instantiation (DUT)
    // ----------------------------------------------------------------
    axi_ram_top #(
        .DATA_WIDTH(DATA_WIDTH),
        .ADDR_WIDTH(ADDR_WIDTH)
    ) dut (
        .clk(aclk),
        .rst_n(aresetn),

        // Write Address
        .s_axi_awaddr(s_axi_awaddr),
        .s_axi_awvalid(s_axi_awvalid),
        .s_axi_awready(s_axi_awready),

        // Write Data
        .s_axi_wdata(s_axi_wdata),
        .s_axi_wstrb(s_axi_wstrb),     // Connected Strobe
        .s_axi_wvalid(s_axi_wvalid),
        .s_axi_wready(s_axi_wready),

        // Write Response
        .s_axi_bresp(s_axi_bresp),
        .s_axi_bvalid(s_axi_bvalid),
        .s_axi_bready(s_axi_bready),

        // Read Address
        .s_axi_araddr(s_axi_araddr),
        .s_axi_arvalid(s_axi_arvalid),
        .s_axi_arready(s_axi_arready),

        // Read Data
        .s_axi_rdata(s_axi_rdata),
        .s_axi_rresp(s_axi_rresp),
        .s_axi_rvalid(s_axi_rvalid),
        .s_axi_rready(s_axi_rready)
    );

    // ----------------------------------------------------------------
    // Clock Generation
    // ----------------------------------------------------------------
    always #5 aclk = ~aclk; // 100 MHz

    // ----------------------------------------------------------------
    // Main Test Sequence
    // ----------------------------------------------------------------
    initial begin
        // Initialize Inputs
        aclk = 0;
        aresetn = 0;
        error_count = 0;

        s_axi_awaddr = 0; s_axi_awvalid = 0;
        s_axi_wdata  = 0; s_axi_wvalid  = 0; s_axi_wstrb = 0;
        s_axi_bready = 0;
        s_axi_araddr = 0; s_axi_arvalid = 0;
        s_axi_rready = 0;

        // Reset Pulse
        #20 aresetn = 1;
        #20;

        $display("\n===================================================");
        $display("STARTING SELF-CHECKING TESTBENCH");
        $display("===================================================");

        // --- TEST CASE 1: Standard Write & Read ---
        axi_write(10'd4, 32'hDEADBEEF);
        axi_read (10'd4, 32'hDEADBEEF);

        // --- TEST CASE 2: Another Address ---
        axi_write(10'd8, 32'hCAFEBABE);
        axi_read (10'd8, 32'hCAFEBABE);

        // --- TEST CASE 3: Check Memory Persistence ---
        // Verify address 4 is still valid after writing to 8
        axi_read (10'd4, 32'hDEADBEEF);

        // --- TEST CASE 4: Read Uninitialized Address (Expect 0) ---
        axi_read (10'd100, 32'h00000000);

        // --- SUMMARY ---
        $display("\n===================================================");
        if (error_count == 0) begin
            $display("SIMULATION PASSED: All checks successful.");
        end else begin
            $display("SIMULATION FAILED: %0d errors found.", error_count);
        end
        $display("===================================================");
        
        $finish;
    end

    // ----------------------------------------------------------------
    // TASK: AXI WRITE
    // ----------------------------------------------------------------
    task axi_write;
        input [ADDR_WIDTH-1:0] addr;
        input [DATA_WIDTH-1:0] data;
        begin
            $display("[WRITE] Addr: 0x%h, Data: 0x%h", addr, data);
            
            @(posedge aclk);
            
            // 1. Assert Address and Data Valids
            s_axi_awaddr  <= addr;
            s_axi_awvalid <= 1;
            
            s_axi_wdata   <= data;
            s_axi_wstrb   <= 4'b1111; // Enable all 4 bytes
            s_axi_wvalid  <= 1;
            
            s_axi_bready  <= 1;       // Ready for response

            // 2. Wait for handshakes (Simple sequential wait for simulation)
            // Note: In a real robust BFM, we would handle these concurrently.
            
        // 2. Wait for handshakes concurrently
fork
    begin
        while (!s_axi_awready) @(posedge aclk);
        s_axi_awvalid <= 0; 
    end
    begin
        while (!s_axi_wready) @(posedge aclk);
        s_axi_wvalid <= 0; 
        s_axi_wstrb  <= 0;
    end
join
           
            
            // 3. Wait for Write Response
            while (!s_axi_bvalid) @(posedge aclk);
            
            // Check Response
            if (s_axi_bresp !== 2'b00) begin
                $display("       [ERROR] Write Response Failed (Resp: %b)", s_axi_bresp);
                error_count = error_count + 1;
            end
            
            s_axi_bready <= 0;        // Transaction Complete
            @(posedge aclk);
        end
    endtask

    // ----------------------------------------------------------------
    // TASK: AXI READ
    // ----------------------------------------------------------------
    task axi_read;
        input [ADDR_WIDTH-1:0] addr;
        input [DATA_WIDTH-1:0] exp_data;
        begin
            @(posedge aclk);
            
            // 1. Assert Read Address
            s_axi_araddr  <= addr;
            s_axi_arvalid <= 1;
            s_axi_rready  <= 1; // We are ready for data immediately
            
            // 2. Wait for Address Acceptance
            while (!s_axi_arready) @(posedge aclk);
            s_axi_arvalid <= 0; // Address Accepted

            // 3. Wait for Read Data Valid
            while (!s_axi_rvalid) @(posedge aclk);
            
            // 4. Verify Data
            if (s_axi_rdata === exp_data) begin
                $display("[READ]  Addr: 0x%h, Data: 0x%h ... OK", addr, s_axi_rdata);
            end else begin
                $display("[READ]  Addr: 0x%h, Data: 0x%h ... MISMATCH (Exp: 0x%h)", addr, s_axi_rdata, exp_data);
                error_count = error_count + 1;
            end
            
            s_axi_rready <= 0; // Transaction Complete
            @(posedge aclk);
        end
    endtask

endmodule
